## Usage

```tsx
import { Separator } from "higgyui";

export function Example() {
  return (
    <div className="grid gap-3">
      <span>Project details</span>
      <Separator />
      <span>Workspace settings</span>
    </div>
  );
}
```

## Props

| Prop | Type | Details |
| --- | --- | --- |
| `orientation` | `"horizontal" or "vertical"` | Defaults to horizontal. Place a vertical separator in a flex container with height. |
| `decorative` | `boolean` | Suppresses separator semantics for purely visual divisions. |
